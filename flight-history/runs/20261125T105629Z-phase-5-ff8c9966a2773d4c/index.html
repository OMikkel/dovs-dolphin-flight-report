record Pair { a: int; b: int; }
// represents the 2x2 matrix
//   a+b b
//    b  a

// everything is computed modulo 10^4

int add(x: int, y: int) { return (x+y) % 10000; }
int mul(x: int, y: int) { return (x*y) % 10000; }

Pair mulp(x : Pair, y : Pair) {
    return new Pair { a = add(mul(x.b,y.b),mul(x.a,y.a));
                      b = add(mul(add(x.a,x.b),y.b), mul(x.b,y.a));};
}

// computes x^n*y for matrix multiplication by fast exponentiation
Pair exp(x: Pair, n: int, y: Pair) {
    if (n == 0) return y;
    var z : Pair = nil;
    if (n % 2 == 1)
        z = mulp(x,y);
    else
        z = y;
    return exp(mulp(x,x), n/2,  z);
}


int fib(n: int) {
    var i = new Pair{ a = 0; b = 1;};
    var p = exp(i, n-1, i);
    return p.b;
}

int main() {
    printInt(fib(10)); 
    printInt(fib(2014)); 
    printInt(fib(1000000000));
    return 0;
}

record Singly { head: int; next: Singly; }

Singly split(l: Singly) {
    var n = 0;
    var l1 = l;
    while (l1 != nil) { l1 = l1.next; n = n+1; };
    n = (n-1)/2;
    var last = l;
    while (n > 0) { last = last.next; n = n-1; };
    var l2 = last.next;
    last.next = nil;
    return l2;
}

Singly merge(a: Singly, b: Singly) {
    if (a == nil) return b;
    if (b == nil) return a;
    if (a.head > b.head) return merge(b, a);
    var res = a; // pointer to the very first element
    var l1: Singly = a;
    var l2: Singly = b;
    // invariant res->...->l1->...->null
    //                     l2->...->null
    //       and l1.head <= l2.head
    while (l2 != nil) {
        var v = l2.head;
        var n = l1.next;
        while (n != nil && n.head <= v) { l1 = n; n = n.next; };
        l1.next = l2;
        l2 = n;
    };
    return res;
}

Singly mergesort(l: Singly) {
    if (l == nil || l.next == nil) return l;
    var l2 = split(l);
    return merge(mergesort(l), mergesort(l2));
}

void print_list(l: Singly) {
    if (l == nil) { output_string("\n", get_stdout()); return; };
    printInt(l.head); 
    print_list(l.next);
}

// Pascal's triangle modulo 7

record List { value: int; next: List; }

int get(l: List, i: int) {
    if (l == nil) return -1; // shouldn't happen
    if (i == 0) return l.value;
    return get(l.next, i-1);
}

void set(l: List, i: int, v: int) {
    if (l == nil) return;
    if (i == 0) l.value = v;
    else set(l.next, i-1, v);
}

void print_row(r: List, i: int) {
    var j = 0;
    while (j <= i) {
      if (get(r, j) != 0)
        output_string("*", get_stdout());
      else
        output_string("0", get_stdout());
      j = j+1;
    };
    output_string("\n", get_stdout());
}

void compute_row(r: List, j: int) {
    var v = 0;
    if (j == 0)
      v = 1;
    else
      v = (get(r, j) + get(r, j-1)) % 7;

int diff(a: int, b: int, c: int) { // a\b\c a&~b&~c
    if (a == 0) return 0;
    var z = 0;
    if (a%2==1 && b%2==0 && c%2==0)
        z = 1;
    else
        z = 0;
    return 2 * diff(a/2, b/2, c/2) + z;
}
int lowest_bit(x: int) {
    if (x == 1) return 1;
    return 2 * lowest_bit(x / 2);
}

int t(a: int, b: int, c: int) {
    var f = 1;
    if (a > 0) {
        f = 0;
        var e = diff(a, b, c);
        while (e > 0) {
            var d = lowest_bit(e);
            f = f + t(a-d, (b+d)*2, (c+d)/2);
            e = e - d;
        }
    };
    return f;
}

int low_bits(n: int) {
    if (n == 0) return 0;
    return 2 * low_bits(n-1) + 1;
}

int q(n: int) {
    return t(low_bits(n), 0, 0);
}

void quicksort(i: int, j: int, a:[int]) {
    if (i >= j - 1) return;
    var p = a[i];
    var pos = i;
    var k = i;
    for (k = k + 1; k < j; k = k + 1) {
        if (a[k] <= p) {
            swap(pos+1, k, a);
            pos = pos + 1;
        };
    };
    swap(i, pos, a);
    quicksort(i, pos, a);
    quicksort(pos+1, j, a);
}

    var k = makepair(0,0);

}

    var i = 1;
